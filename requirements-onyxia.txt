# requirements-onyxia.txt — dependencies installed INSIDE the Onyxia service.
#
# Difference with requirements.txt: no torch, no torchvision. The Datalab image
# already provides a PyTorch paired with its CUDA version; the venv is created
# with --system-site-packages to inherit it. Reinstalling torch here means a
# 2-3 GB download AND a broken CUDA pairing.
#
# We follow the image rather than pin (single exception: numpy, see below).
# The exact versions that produced the figures are in requirements.txt.

# tooling
ipykernel
pytest

# SMC
particles          # test oracle for the resamplers
scipy

# numpy is pinned, and it is the only pinned package. particles 0.4 declares
# numpy<2 in its metadata (stale: resampling.multinomial and .systematic run
# fine on 2.x, checked). Without the pin, pip downgrades the whole venv to
# 1.26.4 while the image's torch is built against numpy 2.x: keep 2.3 to stay
# aligned with the image (2.5.3) without exceeding what numba accepts. pip
# prints a conflict warning about particles: expected.
numpy==2.3.*

# figures and loops
matplotlib
tqdm

# experiment tracking and evaluation
wandb
pytorch-fid
